cmake_minimum_required(VERSION 3.18)
project(adminpanelaz_lws VERSION 0.1.0 LANGUAGES C)

find_package(libwebsockets 4.1 CONFIG REQUIRED)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
find_package(SQLite3 REQUIRED)
find_package(LibArchive REQUIRED)
find_package(ZLIB REQUIRED)
find_package(CURL REQUIRED)
find_package(Threads REQUIRED)

set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS web/index.html)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/web/index.html" dashboard_hex HEX)
string(REGEX REPLACE "(..)" "0x\\1," dashboard_bytes "${dashboard_hex}")
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/dashboard.h" CONTENT
    "static const unsigned char dashboard[] = {@dashboard_bytes@};\n" @ONLY)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS web/portal.html)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/web/portal.html" portal_hex HEX)
string(REGEX REPLACE "(..)" "0x\\1," portal_bytes "${portal_hex}")
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/portal_page.h" CONTENT
    "static const unsigned char portal_page[] = {@portal_bytes@};\n" @ONLY)

file(GLOB web_assets CONFIGURE_DEPENDS RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}/web" "web/*.js" "web/*.css")
set(web_asset_declarations "")
set(web_asset_entries "")
foreach(asset IN LISTS web_assets)
    set(asset_path "${CMAKE_CURRENT_SOURCE_DIR}/web/${asset}")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${asset_path}")
    string(MAKE_C_IDENTIFIER "${asset}" identifier)
    file(READ "${asset_path}" asset_hex HEX)
    string(REGEX REPLACE "(..)" "0x\\1," asset_bytes "${asset_hex}")
    if(asset MATCHES "\\.js$")
        set(type "text/javascript")
    else()
        set(type "text/css")
    endif()
    string(APPEND web_asset_declarations "static const unsigned char asset_${identifier}[] = {${asset_bytes}};\n")
    string(APPEND web_asset_entries "{\"/assets/${asset}\", \"${type}; charset=utf-8\", asset_${identifier}, sizeof(asset_${identifier})},\n")
endforeach()
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/web_assets.h" CONTENT
    "@web_asset_declarations@\nstatic const struct web_asset web_assets[] = {\n@web_asset_entries@{0}};\n" @ONLY)

add_executable(adminpanelaz-lws src/main.c src/host_metrics.c src/vpn_status.c src/wireguard_status.c
    src/panel_auth.c src/panel_backups.c src/panel_backup_create.c src/panel_profiles.c
    src/panel_security.c src/panel_security_totp.c src/panel_client.c src/panel_history.c
    src/panel_nodes.c src/node_transport.c src/panel_settings.c src/panel_portal.c src/panel_ha.c
    src/cidr_routes.c src/cidr_input.c src/panel_cidr_job.c src/panel_telegram.c src/telegram_auth.c)
if(TARGET SQLite3::SQLite3)
    set(sqlite_target SQLite3::SQLite3)
else()
    set(sqlite_target SQLite::SQLite3)
endif()
target_link_libraries(adminpanelaz-lws PRIVATE OpenSSL::Crypto ${sqlite_target} LibArchive::LibArchive ZLIB::ZLIB
    CURL::libcurl Threads::Threads)
if(WIN32)
    target_link_libraries(adminpanelaz-lws PRIVATE ws2_32)
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    find_library(CRYPT_LIBRARY NAMES crypt REQUIRED)
    target_compile_definitions(adminpanelaz-lws PRIVATE PANEL_AUTH_LEGACY_BCRYPT=1)
    target_link_libraries(adminpanelaz-lws PRIVATE ${CRYPT_LIBRARY})
endif()
target_include_directories(adminpanelaz-lws PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_features(adminpanelaz-lws PRIVATE c_std_11)
if(TARGET websockets_shared)
    target_link_libraries(adminpanelaz-lws PRIVATE websockets_shared)
else()
    target_link_libraries(adminpanelaz-lws PRIVATE websockets)
endif()
if(MSVC)
    target_compile_options(adminpanelaz-lws PRIVATE /W4)
else()
    target_compile_options(adminpanelaz-lws PRIVATE -Wall -Wextra -Wpedantic)
endif()
install(TARGETS adminpanelaz-lws RUNTIME DESTINATION bin)
